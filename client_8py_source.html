40 async def check_plan_availability(self, codes: list[str]) -> list[str]:
41 # which of these codes actually return data on the current plan.
42 # api caps a request at 20 codes, so walk it in chunks. sequential
43 # on purpose - hammering it in parallel just trips the rate limit.
44 CHUNK = 20
45 available: list[str] = []
46
47 for i in range(0, len(codes), CHUNK):
48 chunk = codes[i:i + CHUNK]
49 url = f"{BASE_URL}/prices/latest?by_code={','.join(chunk)}"
50 try:
51 async with httpx.AsyncClient(timeout=15) as http:
52 r = await http.get(url, headers=self.headers)
53 r.raise_for_status()
54 body = r.json()
55 data = body.get("data", {})
56 if isinstance(data, dict) and "prices" in data:
57 available.extend(
58 item["code"] for item in data["prices"] if item.get("code")

60 elif isinstance(data, dict) and "price" in data and data.get("code"):
61 available.append(data["code"])
62 except Exception as exc:
63 print(f"plan-check: chunk {i//CHUNK + 1} failed: {exc}")
64

95 async def fetch_prices(
96 self,
97 codes: list[str],
98 known_solo: set[str] | None = None,
99 ) -> tuple[list[dict], set[str]]:
100 # latest prices for a pile of codes, using as few api calls as possible:
101 # round 1 batches of 20
102 # round 2 whatever dropped out, retried in groups of 10
103 # round 3 still missing? one at a time
104 # codes in known_solo skip straight to round 3. we hand back the set of
105 # codes that only worked solo this run so the caller can remember them
106 # and skip the wasted batch attempts next time.
107 CHUNK = 20
108 RETRY_CHUNK = 10
109 known_solo = known_solo or set()
110
111 results: list[dict] = []
112 need_retry: list[str] = [] # dropped from the round-1 batches
113 solo_fetch: list[str] = list(known_solo) # known solo + whatever we discover
114 newly_solo: set[str] = set()
115
116 # ---- Round 1: batches of 20, skipping known-solo codes ----
117 batch_codes = [c for c in codes if c not in known_solo]
118 for i in range(0, len(batch_codes), CHUNK):
119 chunk = batch_codes[i:i + CHUNK]
120 returned = await self._fetch_chunk(chunk, f"batch chunk {i // CHUNK + 1}")
121 results.extend(returned.values())
122 dropped = [c for c in chunk if c not in returned]
123 if dropped:
124 need_retry.extend(dropped)
125
126 # ---- Round 2: retry the dropped ones in groups of 10 ----
127 if need_retry:
128 print(f"fetch_prices: {len(need_retry)} dropped from batches, retrying in groups of {RETRY_CHUNK}")
129 still_missing: list[str] = []
130 for i in range(0, len(need_retry), RETRY_CHUNK):
131 chunk = need_retry[i:i + RETRY_CHUNK]
132 returned = await self._fetch_chunk(chunk, f"retry group {i // RETRY_CHUNK + 1}")
133 results.extend(returned.values())
134 dropped = [c for c in chunk if c not in returned]
135 if dropped:
136 still_missing.extend(dropped)
137 if still_missing:
138 print(f"fetch_prices: {len(still_missing)} still missing, going solo")
139 solo_fetch.extend(c for c in still_missing if c not in known_solo)
140 newly_solo.update(still_missing)
141
142 # ---- Round 3: solo fetches ----
143 for code in solo_fetch:
144 url = f"{BASE_URL}/prices/latest?by_code={code}"
145 try:
146 async with httpx.AsyncClient(timeout=10) as http:
147 r = await http.get(url, headers=self.headers)
148 r.raise_for_status()
149 body = r.json()
150 item = body.get("data", {})
151 if isinstance(item, dict) and item.get("price") is not None:
152 results.append({
153 "code": item.get("code", code),
154 "price": float(item["price"]),
155 "currency": item.get("currency", "USD"),
156 "fetched_at": _normalize_timestamp(item.get("created_at")),
157 })
158 print(f"fetch_prices: solo ok {code}")
159 else:
160 print(f"fetch_prices: solo no data {code}")
161 except Exception as exc:
162 print(f"fetch_prices: solo failed {code}: {exc}")
163
164 return results, newly_solo

168 async def _fetch_chunk(self, codes: list[str], label: str) -> dict[str, dict]:
169 # fetch one batch, return {code: normalized dict} for whatever came back
170 url = f"{BASE_URL}/prices/latest?by_code={','.join(codes)}"
171 try:
172 async with httpx.AsyncClient(timeout=10) as http:
173 r = await http.get(url, headers=self.headers)
174 r.raise_for_status()
175 body = r.json()
176 except Exception as exc:
177 print(f"fetch_prices: {label} failed: {exc}")
178 return {}
179
180 data = body.get("data", {})
181 if isinstance(data, dict) and "prices" in data:
182 raw = data["prices"]
183 elif isinstance(data, dict) and "price" in data:
184 raw = [data]
185 else:
186 return {}
187
188 out: dict[str, dict] = {}
189 for item in raw:
190 if isinstance(item, dict) and item.get("price") is not None:
191 code = item.get("code", codes[0])
192 out[code] = {
193 "code": code,
194 "price": float(item["price"]),
195 "currency": item.get("currency", "USD"),
196 "fetched_at": _normalize_timestamp(item.get("created_at")),
197 }
198 return out

211def _normalize_timestamp(ts: str | None) -> str:
212 # pin an api timestamp to iso-8601 utc. if it's missing or junk, use now.
213 if not ts:
214 return _now_utc()
215 try:
216 dt = datetime.fromisoformat(ts.replace("Z", "+00:00"))
217 return dt.strftime("%Y-%m-%dT%H:%M:%SZ")
218 except (ValueError, AttributeError):
219 return _now_utc()
