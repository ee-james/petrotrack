135/** Return metadata for a code, with sensible fallback. */
136export function getMeta(code) {
137 if (meta[code]) return meta[code];
138 const name = code
139 .replace(/_(USD|EUR|GBP|NOK)$/, '')
140 .replace(/_/g, ' ')
141 .toLowerCase()
142 .replace(/\b\w/g, c => c.toUpperCase());
143 return { cc: null, flag: '🌐', name, desc: null };
144}
