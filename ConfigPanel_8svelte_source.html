37 async function initPanel() {
38 saveError = null;
39 themeError = null;
40 await Promise.all([initCommodities(), initTheme()]);
41 }

43 async function initCommodities() {
44 loadingComm = true;
45 selected = [...currentEnabled];
46 filter = '';
47 try {
48 const res = await getAvailableCommodities();
49 // Always include currently enabled codes even if not in available list
50 available = [...new Set([...res.codes, ...currentEnabled])];
51 } catch (e) {
52 saveError = e.message;
53 } finally {
54 loadingComm = false;
55 }
56 }
57
58 async function initTheme() {
59 loadingTheme = true;
60 try {
61 const cfg = await getConfig();
62 // Deep-copy so edits don't affect the live config object
63 colors = { ...cfg.theme.colors };
64 motion = { ...cfg.theme.motion };
65 tiles = { ...cfg.theme.tiles };
66 } catch (e) {
67 themeError = e.message;
68 } finally {
69 loadingTheme = false;
70 }
71 }
72
73 // ---- Commodities actions ----
74
75 async function refreshCatalog() {
76 refreshingCat = true;
77 saveError = null;
78 try {
79 // Fetch catalog and plan availability concurrently
80 const [catRes, planRes] = await Promise.all([
81 getCommoditiesCatalog(),
82 getPlanCheck(),
83 ]);
84 // Merge: catalog + currently selected so pre-selected items never vanish
85 available = [...new Set([...catRes.codes, ...selected])];
86 unavailableCodes = new Set(planRes.unavailable);
87 } catch (e) {
88 saveError = e.message;
89 } finally {
90 refreshingCat = false;
91 }
92 }
93
94 function toggle(code) {
95 if (unavailableCodes?.has(code)) return; // not on plan - non-interactive
96 if (selected.includes(code)) {
97 if (selected.length > 1) selected = selected.filter(c => c !== code);
98 } else {
99 selected = [...selected, code];
100 }
101 }
102
103 function saveCommodities() {
104 if (selected.length < 1) return;
105 dispatch('save', { enabled: selected });
106 }
107
108 // ---- Theme actions ----
109
110 async function saveTheme() {
111 themeSaving = true;
112 themeError = null;
113 try {
114 await putTheme({ colors, motion, tiles });
115 dispatch('save-theme');
116 } catch (e) {
117 themeError = e.message;
118 } finally {
119 themeSaving = false;
120 }
121 }
122
123 function close() {
124 dispatch('close');
125 }
