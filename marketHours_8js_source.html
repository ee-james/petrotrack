62/** Convert a UTC Date to Eastern Time components. */
63function toET(date) {
64 // toLocaleString returns ET wall-clock time as a string;
65 // wrapping in new Date() gives us a Date whose .getDay() / .getHours() etc.
66 // reflect ET (treating it as local - a well-known JS timezone hack).
67 return new Date(date.toLocaleString('en-US', { timeZone: 'America/New_York' }));
68}

70function formatDuration(ms) {
71 const totalMin = Math.ceil(ms / 60_000);
72 const h = Math.floor(totalMin / 60);
73 const m = totalMin % 60;
74 if (h === 0) return `${m}m`;
75 if (m === 0) return `${h}h`;
76 return `${h}h ${m}m`;
77}

81function f245Status(now) {
82 const et = toET(now);
83 const day = et.getDay(); // 0=Sun … 6=Sat
84 const t = et.getHours() + et.getMinutes() / 60; // decimal hours in ET
85
86 // Determine open/closed
87 let open;
88 if (day === 6) open = false; // Saturday
89 else if (day === 0) open = t >= 18; // Sunday: opens 18:00
90 else if (day === 5) open = t < 17; // Friday: closes 17:00
91 else /* Mon-Thu */ open = !(t >= 17 && t < 18); // daily break 17-18
92
93 if (open) return { open: true };
94
95 // Compute hours until next open
96 let h;
97 if (day === 6) h = (24 - t) + 18; // → Sun 18:00
98 else if (day === 5 && t >= 17) h = (24 - t) + 24 + 18; // → Sun 18:00 (+Sat)
99 else if (day === 0 && t < 18) h = 18 - t; // → today 18:00
100 else /* daily break */ h = 18 - t; // → today 18:00
101
102 return { open: false, opensIn: formatDuration(h * 3_600_000) };
103}

107/**
108 * Returns null (no schedule) or { open: boolean, opensIn?: string }.
109 * opensIn is a human-readable string like "2h 30m" when open === false.
110 */
111export function getMarketStatus(code) {
112 const schedule = SCHEDULE[code] ?? null;
113 if (!schedule) return null;
114 if (schedule === 'f245') return f245Status(new Date());
115 return null;
