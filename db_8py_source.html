11def init_db(path: str) -> None:
12 # tables + index, only if they aren't there already
13 dir_path = os.path.dirname(path)
14 if dir_path:
15 os.makedirs(dir_path, exist_ok=True)
16
17 conn = sqlite3.connect(path)
18 try:
19 conn.executescript("""
20 CREATE TABLE IF NOT EXISTS prices (
21 id INTEGER PRIMARY KEY AUTOINCREMENT,
22 code TEXT NOT NULL,
23 price REAL NOT NULL,
24 currency TEXT NOT NULL DEFAULT 'USD',
25 fetched_at TEXT NOT NULL
26 );
27 CREATE INDEX IF NOT EXISTS idx_prices_code_fetched
28 ON prices(code, fetched_at DESC);
29
30 CREATE TABLE IF NOT EXISTS settings (
31 code TEXT NOT NULL,
32 key TEXT NOT NULL,
33 value TEXT NOT NULL,
34 PRIMARY KEY (code, key)
35 );
36 """)
37 conn.commit()
38 finally:
39 conn.close()

61def query_prices(
62 conn: sqlite3.Connection,
63 code: str,
64 since: datetime,
65) -> list[dict]:
66 # every price for this code since `since` (utc), newest first
67 since_str = since.strftime("%Y-%m-%dT%H:%M:%SZ")
68 cursor = conn.execute(
69 "SELECT price, currency, fetched_at FROM prices "
70 "WHERE code = ? AND fetched_at >= ? ORDER BY fetched_at DESC",
71 (code, since_str),
72 )
73 return [dict(row) for row in cursor.fetchall()]

90def get_recent_change_timestamps(
91 conn: sqlite3.Connection, code: str, n: int = 10
92) -> list[str]:
93 # fetched_at for the last N rows where the price actually moved. feeds the
94 # adaptive scheduler - average the gaps between these and you know roughly
95 # how often this commodity is worth re-checking.
96 cursor = conn.execute(
97 """
98 SELECT fetched_at, price FROM prices
99 WHERE code = ?
100 ORDER BY fetched_at DESC
101 LIMIT 200
102 """,
103 (code,),
104 )
105 rows = cursor.fetchall()
106 change_times: list[str] = []
107 prev_price = None
108 for row in rows:
109 if prev_price is not None and row["price"] != prev_price:
110 change_times.append(row["fetched_at"])
111 if len(change_times) >= n:
112 break
113 prev_price = row["price"]
114 return change_times
