29 function nyParts(date) {
30 const parts = new Intl.DateTimeFormat('en-US', {
31 timeZone: 'America/New_York',
32 weekday: 'short', hour: 'numeric', minute: '2-digit', hour12: false,
33 }).formatToParts(date);
34 const get = t => parts.find(p => p.type === t)?.value ?? '0';
35 const h = parseInt(get('hour'));
36 const m = parseInt(get('minute'));
37 return { wd: get('weekday'), h, m, t: h * 60 + m };
38 }

40 function fmtNY(date) {

43 weekday: 'short', hour: 'numeric', minute: '2-digit', hour12: true,
44 }).format(date);
45 }

52 }

54 // CME Globex Energy & Metals (WTI, NatGas, Gasoline, Heating Oil, Gold)
55 // Sun 18:00 ET -> Fri 17:00 ET; daily maintenance break 17:00-18:00 ET
56 function isCMEOpen({ wd, t }) {
57 if (wd === 'Sat') return false;
58 if (wd === 'Sun') return t >= 18 * 60;
59 if (wd === 'Fri') return t < 17 * 60;
60 return !(t >= 17 * 60 && t < 18 * 60);
61 }
62
63 // ICE Futures Europe (Brent Crude, Gasoil, EU/UK Carbon)
64 // Opens later Sunday; Mon-Thu break 17:00-19:00 ET
65 function isICEOpen({ wd, t }) {
66 if (wd === 'Sat') return false;
67 if (wd === 'Sun') return t >= 20 * 60;
68 if (wd === 'Fri') return t < 17 * 60;
69 return !(t >= 17 * 60 && t < 19 * 60);
70 }
71
72 // Forex OTC (EUR/USD, GBP/USD, USD/NOK, EUR/NOK)
73 // Sun 17:00 ET -> Fri 17:00 ET, continuous
74 function isForexOpen({ wd, t }) {
75 if (wd === 'Sat') return false;
76 if (wd === 'Sun') return t >= 17 * 60;
77 if (wd === 'Fri') return t < 17 * 60;
78 return true;
79 }
80
81 // OTC Coal & LNG reference prices - business hours only
82 // Mon-Fri 08:00-17:00 ET
83 function isOTCOpen({ wd, t }) {
84 if (wd === 'Sat' || wd === 'Sun') return false;
85 return t >= 8 * 60 && t < 17 * 60;
86 }
